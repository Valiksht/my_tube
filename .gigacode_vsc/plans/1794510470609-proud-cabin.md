# План исправления кнопок лайков, подписок и комментариев

## Проблема

Кнопки лайков и подписок на странице видео выводят ошибки:
- «Не удалось обработать лайк»
- «Не удалось обновить подписку»

Комментарии отправляются, но механизм требует проверки.

## Причина

**Дублирование обработчиков событий в JavaScript.**

На странице `/video/{id}` есть ДВА набора обработчиков для `.like-btn` и `.subscribe-btn`:

1. **Inline-скрипт в `video.html`** (строки 314-390) — прикрепляет обработчики при `DOMContentLoaded`
2. **Глобальный `app.js`** (функция `initFormHandlers()`) — также прикрепляет обработчики для всех кнопок

При клике срабатывают ОБА обработчика почти одновременно, отправляя ДВА fetch-запроса к одному эндпоинту:
- Первый запрос succeeds → состояние переключается
- Второй запрос либо fails (409 конфликт), либо переключает обратно
- Второй запрос попадает в `.catch()` → показывается ошибка

Комментарии работают через обычную HTML-форму POST (не AJAX), поэтому дублирования нет. Но форма POSTит на JSON-API эндпоинт, что некорректно для HTML-формы.

## Исправления

### 1. Удалить inline-скрипт из `video.html`

**Файл:** `app/templates/video.html`, строки 313-391 (`{% block extra_js %}`)

Удалить весь блок `{% block extra_js %}...{% endblock %}`.

**Почему:** Глобальный `app.js` уже содержит обработчики для `.like-btn` и `.subscribe-btn`. Inline-скрипт дублирует эту логику и вызывает двойные запросы.

**Важно:** Глобальный обработчик в `app.js` НЕ переключает класс `.active`. Это нужно исправить.

### 2. Добавить переключение `.active` класса в `app.js`

**Файл:** `app/static/js/app.js`

#### A) Like button handler (строки 68-99)

Добавить переключение класса `.active` после успешного запроса:

```javascript
// Like кнопки
document.querySelectorAll('.like-btn').forEach(function(btn) {
    btn.addEventListener('click', function() {
        var videoId = this.dataset.videoId;
        var countEl = this.querySelector('.like-count');
        
        fetch('/api/videos/' + videoId + '/likes', {
            method: 'POST',
            credentials: 'include',
            headers: { 'X-CSRF-Token': getCSRFToken() }
        })
        .then(function(response) {
            if (response.ok) {
                // Переключить класс active
                this.classList.toggle('active');
                // Обновить счётчик
                return fetch('/api/videos/' + videoId + '/likes/count');
            }
            throw new Error('Like failed');
        }.bind(this))  // <-- bind(this) для доступа к btn в then
        .then(function(response) {
            return response.json();
        })
        .then(function(data) {
            if (countEl) {
                countEl.textContent = formatNumber(data.like_count);
            }
        })
        .catch(function(error) {
            console.error('Error:', error);
            showAlert('Не удалось обработать лайк', 'error');
        });
    });
});
```

Ключевые изменения:
- `.bind(this)` в первом `.then()` — чтобы `this` ссылался на кнопку
- `this.classList.toggle('active')` — переключает визуальное состояние

```javascript
// Subscribe кнопки
document.querySelectorAll('.subscribe-btn').forEach(function(btn) {
    btn.addEventListener('click', function() {
        var channelId = this.dataset.channelId;
        var isSubscribed = this.classList.contains('active');
        
        fetch('/api/channels/' + channelId + '/subscriptions', {

        .then(function(response) {

                var text = btn.textContent.trim();

                            countEl.textContent = data.subscriber_count + ' подписчиков';
                        });
                }
            }
            throw new Error('Subscribe failed');
        })
        .catch(function(error) {
            console.error('Error:', error);
            showAlert('Не удалось обновить подписку', 'error');
        });
    });
```

Добавить обработчик в `app.js` (в функцию `initFormHandlers`):

```javascript
// Comment form
var commentForm = document.getElementById('comment-form');
if (commentForm) {
    commentForm.addEventListener('submit', function(e) {
        e.preventDefault();
        
        var contentEl = document.getElementById('comment-content');
        var errorEl = document.getElementById('comment-form-error');
        var successEl = document.getElementById('comment-form-success');
        var content = contentEl.value.trim();
        
        if (!content) {
            return;
        }
        
        // Получить video_id из action или data-атрибута
        var videoId = commentForm.dataset.videoId;
        
        fetch('/api/videos/' + videoId + '/comments', {
            method: 'POST',
            credentials: 'include',
            headers: {
                'Content-Type': 'application/json',
                'X-CSRF-Token': getCSRFToken()
            },
            body: JSON.stringify({ content: content })
        })
        .then(function(response) {
            if (response.ok) {
                return response.json();
            }
            throw new Error('Comment failed');
        })
        .then(function(data) {
            // Очистить поле
            contentEl.value = '';
            
            // Показать сообщение об успехе
            if (successEl) {
                successEl.textContent = 'Комментарий добавлен';
                successEl.style.display = 'block';
                setTimeout(function() { successEl.style.display = 'none'; }, 3000);
            }
            
            // Перезагрузить комментарии
            return fetch('/api/videos/' + videoId + '/comments?skip=0&limit=50');
        })
        .then(function(response) { return response.json(); })
        .then(function(comments) {
            // Перерисовать комментарии
            renderComments(comments);
        })
        .catch(function(error) {
            console.error('Error:', error);
            if (errorEl) {
                errorEl.textContent = 'Не удалось отправить комментарий';
                errorEl.style.display = 'block';
                setTimeout(function() { errorEl.style.display = 'none'; }, 5000);
            }
        });
    });
}
```

Добавить функцию для рендеринга комментариев:

```javascript
// Рендер комментариев
function renderComments(comments) {
    var section = document.querySelector('.comments-section');
    if (!section) return;
    
    var list = section.querySelector('.comment-list') || createCommentList(section);
    list.innerHTML = '';
    
    for (var i = 0; i < comments.length; i++) {
        var comment = comments[i];
        var div = document.createElement('div');
        div.className = 'comment';
        div.innerHTML = 
            '<div class="comment-header">' +
                '<span class="comment-author">' + escapeHtml(comment.user.username) + '</span>' +
                '<span class="comment-date">' + comment.created_at + '</span>' +
            '</div>' +
            '<p class="comment-content">' + escapeHtml(comment.content) + '</p>';
        list.appendChild(div);
    }
    
    // Обновить счётчик
    var heading = section.querySelector('h2');
    if (heading) {
        heading.textContent = 'Комментарии (' + comments.length + ')';
    }
}

function createCommentList(section) {
    var list = document.createElement('div');
    list.className = 'comment-list';
    
    // Вставить после заголовка h2
    var h2 = section.querySelector('h2');
    if (h2) {
        section.insertBefore(list, h2.nextSibling);
    } else {
        section.appendChild(list);
    }
    
    return list;
}

function escapeHtml(text) {
    var div = document.createElement('div');
    div.appendChild(document.createTextNode(text));
    return div.innerHTML;
}
```
